VISA Kubernetes Controller has been deployed successfully.

Release:   {{ .Release.Name }}
Namespace: {{ .Release.Namespace }}

{{- if .Values.controller.enabled }}
---
VISA Kubernetes Controller:

1. Check the controller pod status:
   kubectl get pods -n {{ include "visa-k8s-controller.namespace" . }} -l "app.kubernetes.io/name={{ include "visa-k8s-controller.name" . }},app.kubernetes.io/instance={{ .Release.Name }}"

2. View controller manager logs:
   kubectl logs -n {{ include "visa-k8s-controller.namespace" . }} -l "app.kubernetes.io/name={{ include "visa-k8s-controller.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -c manager -f

3. Verify VISA CRDs:
   kubectl get crds | grep 'visa.k8s'

{{- if .Values.metrics.enabled }}
4. Access controller metrics:
   kubectl port-forward -n {{ include "visa-k8s-controller.namespace" . }} svc/{{ include "visa-k8s-controller.fullname" . }} {{ .Values.service.port }}:{{ .Values.service.port }}
   {{- if .Values.metrics.secure }}
   curl -k https://127.0.0.1:{{ .Values.service.port }}/metrics -H "Authorization: Bearer $(kubectl create token {{ include "visa-k8s-controller.serviceAccountName" . }} -n {{ include "visa-k8s-controller.namespace" . }})"
   {{- else }}
   curl http://127.0.0.1:{{ .Values.service.port }}/metrics
   {{- end }}
{{- end }}
{{- end }}

{{- if .Values.webCloudProvider.enabled }}
---
VISA Web Cloud Provider:

1. Check the web cloud provider pod status:
   kubectl get pods -n {{ include "visa-k8s-web-cloud-provider.namespace" . }} -l "app.kubernetes.io/name={{ include "visa-k8s-web-cloud-provider.name" . }},app.kubernetes.io/instance={{ .Release.Name }}"

2. View web cloud provider logs:
   kubectl logs -n {{ include "visa-k8s-web-cloud-provider.namespace" . }} -l "app.kubernetes.io/name={{ include "visa-k8s-web-cloud-provider.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -f

3. Retrieve the generated Auth API Key from the ConfigMap:
   kubectl get configmap {{ include "visa-k8s-web-cloud-provider.configMapName" . }} -n {{ include "visa-k8s-web-cloud-provider.namespace" . }} -o jsonpath="{.data.AUTH_API_KEY}"

4. Access the web cloud provider API:
   kubectl port-forward -n {{ include "visa-k8s-web-cloud-provider.namespace" . }} svc/{{ include "visa-k8s-web-cloud-provider.fullname" . }} {{ .Values.webCloudProvider.service.port }}:{{ .Values.webCloudProvider.service.port }}
   curl http://127.0.0.1:{{ .Values.webCloudProvider.service.port }}/healthz
{{- end }}
